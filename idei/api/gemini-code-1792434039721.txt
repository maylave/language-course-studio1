<!-- src/components/Popular.vue -->
<script setup lang="ts">
import { ref, onMounted } from 'vue'
import CourseCard from '@/components/course/CourseCard.vue'
import { learningApi } from '@/api'
import type { CourseItem } from '@/types/learning'

const coursesList = ref<CourseItem[]>([])
const loading = ref(false)

const loadCourses = async (category?: string) => {
  loading.value = true
  try {
    coursesList.value = await learningApi.getCourses(category)
  } finally {
    loading.value = false
  }
}

onMounted(() => {
  loadCourses()
})
</script>

<template>
  <!-- В сетке рендерим coursesList -->
  <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-6">
    <CourseCard
      v-for="course in coursesList"
      :key="course.id"
      :title="course.title"
      :description="course.description"
      :icon-emoji="course.iconEmoji"
      :xp="course.xp"
      :badge="course.badge"
      :levels="course.levels"
      :students="course.students"
      :loading="loading"
    />
  </div>
</template>
